export const meta = {
  name: 'curation-coherence-pass',
  description: 'Judge every order inversion per season, research off-platform dates and animation placements, refute each proposal adversarially',
  phases: [
    { title: 'Order', detail: 'one judge per season doc with inversions' },
    { title: 'Research', detail: 'DHMIS, Critical Role, Hertzfeldt, animated shorts' },
    { title: 'Verify', detail: 'two skeptics per proposal, independent sources per research result' },
  ],
}

const INV = args.inversionsPath
const VERDICT = {
  type: 'object',
  properties: {
    doc: { type: 'string' },
    pattern: { type: 'string', description: 'any ordering pattern the doc follows, e.g. talkbacks follow their episode; empty if none' },
    verdicts: {
      type: 'array',
      items: {
        type: 'object',
        properties: {
          earlier_in_file: { type: 'string' },
          later_in_file: { type: 'string' },
          verdict: { type: 'string', enum: ['convention', 'documented', 'reorder', 'date-suspect', 'unsure'] },
          evidence: { type: 'string' },
          proposal: { type: 'string', description: 'for reorder: "move X to after Y" / "move X to before Y"; for date-suspect: which date, probable value, source; else empty' },
        },
        required: ['earlier_in_file', 'later_in_file', 'verdict', 'evidence', 'proposal'],
      },
    },
  },
  required: ['doc', 'pattern', 'verdicts'],
}
const REFUTE = {
  type: 'object',
  properties: {
    refuted: { type: 'boolean' },
    reasoning: { type: 'string' },
    correction: { type: 'string', description: 'if refuted or partly wrong, what is right instead; else empty' },
  },
  required: ['refuted', 'reasoning', 'correction'],
}

const RULES = `The rule being checked: within one doc of curation/seasons.yaml, entries are in release order by \`published:\`. Deliberate exceptions, and the only ones: (1) a trailer may stand ahead of what it trails whatever its date -- a season's opening trailers lead the doc, and a trailer for one special or episode sits right before it; (2) episodes keep Dropout's episode-number order even where that is not release order (the number is Dropout's, the date is the release, and the two need not agree -- Breaking News season 2 is the known case); (3) \`animation:\` entries sit beside the moment they depict and are out of order by design (they are already left out of the list you are given); (4) a comment in the file may record a deliberate exception, in which case quote it. Anything else that is out of release order is either a mistake in the file order or a wrong date, and you must say which.

Sources: read the whole doc in curation/seasons.yaml, including the comment block above its \`show:\` line and comments between entries (find it with Grep for the exact \`season:\` and \`show:\` lines, then Read that range). For talkbacks (\`talkback: true\`) and anything with a \`dropout:\` slug, data/dropout.yaml has a record per slug with \`description\`, \`releaseDate\`, \`episodeNumber\` and \`collection\`; a talkback's description usually names the episode the cast discusses. \`published:\` is meant to equal Dropout's \`releaseDate\` where the page carries one (deno task verify-dates enforces this), so a wrong-looking date is a claim that Dropout's page is wrong and needs independent evidence: the YouTube upload date in data/videos/<channel>.yaml (field publishedAt; for a members upload that is the upload, which can precede release by days or weeks, so it is a lower bound on nothing and an upper bound on the release only loosely), the show's weekly cadence, or a free upload. Do not propose a reorder that would put an episode out of its number order. Do not propose moving an entry to a position that merely creates a different inversion. Be concrete: name the entries exactly as they appear in the file.`

const judge = (doc) => agent(
  `You are judging the order of entries in one season doc: "${doc.key}" (show "${doc.show}", season ${doc.season === null ? 'none -- a season-less doc' : JSON.stringify(doc.season)}).

${RULES}

The inversions to judge are in the JSON file ${INV}: read it and take the object whose "key" is ${JSON.stringify(doc.key)}. Each inversion is an adjacent pair where the later entry in the file has an earlier date. For every one, return a verdict: convention (say which of rules 1-3), documented (quote the comment), reorder (say exactly where the later entry should move, as "move <entry> to after <entry>" or "move <entry> to before <entry>", and why that position is right by date and by sense), date-suspect (which date is probably wrong, what it probably is, and the evidence), or unsure (what would settle it). Where several adjacent inversions are one displaced block (for example a run of cast behind-the-scenes videos all dated before the finale they follow), give the whole block one proposal and repeat it on each row. Also report any ordering pattern this doc follows that the rest of the catalogue should be checked against.`,
  { label: `order:${doc.key}`, phase: 'Order', schema: VERDICT },
)

const skeptics = (verdict, row) => parallel([
  () => agent(`A proposal has been made about curation/seasons.yaml and your job is to try to refute it on the evidence. Doc: ${JSON.stringify(verdict.doc)}. Pair: earlier in file ${JSON.stringify(row.earlier_in_file)}; later in file ${JSON.stringify(row.later_in_file)}. Verdict: ${row.verdict}. Proposal: ${JSON.stringify(row.proposal)}. Claimed evidence: ${JSON.stringify(row.evidence)}.

${RULES}

Check the evidence yourself: open the doc, the data/dropout.yaml records, the upload dates. Does the proposal follow from the facts? Would it leave the doc in release order (apart from the stated exceptions) rather than create a new inversion? Is the cited description or date actually what the file says? If any part fails, refuted=true and say what is right instead. If you are uncertain, refuted=true.`, { label: `refute-facts:${row.later_in_file.slice(0, 40)}`, phase: 'Verify', schema: REFUTE }),
  () => agent(`A proposal has been made about curation/seasons.yaml and your job is to judge it against the catalogue's own conventions, as a skeptic. Doc: ${JSON.stringify(verdict.doc)}. Pair: earlier in file ${JSON.stringify(row.earlier_in_file)}; later in file ${JSON.stringify(row.later_in_file)}. Verdict: ${row.verdict}. Proposal: ${JSON.stringify(row.proposal)}. Claimed evidence: ${JSON.stringify(row.evidence)}.

${RULES}

Find how the rest of curation/seasons.yaml orders the same situation (the same kind of extra in another season of the same show; a talkback relative to its episode; a cast behind-the-scenes set; a post-season extra) and whether this proposal agrees with that practice or with a comment somewhere in the file or in CLAUDE.md / dates.md that settles it. If the proposal would make this doc inconsistent with the rest, or contradicts a documented decision, refuted=true and say what the consistent placement is. If you are uncertain, refuted=true.`, { label: `refute-convention:${row.later_in_file.slice(0, 40)}`, phase: 'Verify', schema: REFUTE }),
]).then((votes) => ({ ...row, doc: verdict.doc, votes: votes.filter(Boolean) }))

const orderResults = pipeline(
  args.docs,
  (doc) => judge(doc),
  async (verdict) => {
    if (!verdict) return null
    const contested = verdict.verdicts.filter((r) => r.verdict === 'reorder' || r.verdict === 'date-suspect')
    const checked = await parallel(contested.map((row) => () => skeptics(verdict, row)))
    return { ...verdict, checked: checked.filter(Boolean) }
  },
)

const RESEARCH_DISABLED = () => pipeline(
  Object.entries(RESEARCH),
  ([key, r]) => agent(r.prompt, { label: r.label, phase: 'Research', schema: r.schema }).then((result) => ({ key, result })),
  async ({ key, result }) => {
    if (!result) return { key, result: null, check: null }
    const check = await agent(`Another researcher produced the following result. Your job is to try to refute it, row by row, from sources you find yourself (WebSearch and WebFetch are available; prefer a different source from the one each row cites). Original task: ${RESEARCH[key].prompt.slice(0, 1500)}

Result to check:
${JSON.stringify(result, null, 1)}

For every row say whether it holds, and where it does not, what the right value is and your source. Then set refuted=true if any row is wrong or unsupported, with the corrections listed in "correction" as one line per row ("<title>: <field> should be <value> (<source>)"). If every row holds, refuted=false.`, { label: `refute:${key}`, phase: 'Verify', schema: REFUTE })
    return { key, result, check }
  },
)

const order = await orderResults
